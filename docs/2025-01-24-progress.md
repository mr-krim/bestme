## Technical Notes

### ONNX Runtime Pattern
```rust
// Correct pattern for ONNX Runtime 1.16
let input_cow = CowArray::from(input_array.into_dyn());
let outputs = {
    let input_value = Value::from_array(
        self.session.allocator(),
        &input_cow
    )?;
    self.session.run(vec![input_value])?
};
```

### OpenTelemetry 0.24 Pattern
```rust
// No Context parameter needed
self.inference_count.add(1, &labels);
// String lifetimes
span.set_attribute(KeyValue::new(key.to_string(), value.into()));
```

```rust
let mut search_attributes = HashMap::new();
search_attributes.insert("application", SERVICE_NAME);
collection.search_items(search_attributes).await?;
```
